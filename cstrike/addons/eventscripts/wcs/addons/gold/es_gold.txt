event player_activate
{
  es_exists gold_exists key gold event_var(es_steamid)
  es_exists ex_1 key golddata event_var(es_steamid)
  ifx false(gold_exists) do
  {
    es_keycreate gold event_var(es_steamid)
    es_keysetvalue gold event_var(es_steamid) gold 0
    es_xkeygroupsave gold |gold
  }
  ifx false(ex_1) do
  {
    es_keycreate golddata event_var(es_steamid)
    es_keysetvalue golddata event_var(es_steamid) gold 0
  }
}

event player_death
{
    es_exists wcs_tmp userid event_var(attacker)
    if (server_var(wcs_tmp) = 1) do
    {  
	   if (event_var(attacker) != event_var(userid)) do
	   {
      es_xkeygetvalue gold_rand gold_config gold_settings gold_per_kill
	    es_xkeygetvalue gold_headshot gold_config gold_settings headshot
	    es_xkeygetvalue gold_knife gold_config gold_settings knife
		  es_keygetvalue curgold gold event_var(es_attackersteamid) gold
		  es_keygetvalue gold_kills gold event_var(es_attackersteamid) kills
		  es_keygetvalue gold_deaths gold event_var(es_steamid) deaths
		  if (event_var(headshot) == 1) do 
      { 
        es_math curgold + server_var(gold_headshot)
        es_keysetvalue gold event_var(es_attackersteamid) gold server_var(curgold)
        es_format gold_format "You Gained %1 Additional Gold for a HeadShot!" server_var(gold_headshot)
        es_delayed 1 es usermsg hudhint event_var(attacker) server_var(gold_format)
      }
      if (event_var(weapon) == knife) do 
      {
        es_math curgold + server_var(gold_knife)
        es_keysetvalue gold event_var(es_attackersteamid) gold server_var(curgold)
        es_format gold_format "You Gained %1 Additional Gold for a Knife Kill!!" server_var(gold_knife)
        es_delayed 1 es usermsg hudhint event_var(attacker) server_var(gold_format)
      }
		  es_math curgold + server_var(gold_rand)
		  es_xmath gold_kills + 1
		  es_xmath gold_deaths + 1
      es_keysetvalue gold event_var(es_attackersteamid) gold server_var(curgold)
      es_keysetvalue gold event_var(es_attackersteamid) kills server_var(gold_kills)
      es_keysetvalue gold event_var(es_steamid) deaths server_var(gold_deaths)
      es_format gold_format "You Gained %1 Gold For killing an enemy" server_var(gold_rand)
		  es usermsg hudhint event_var(attacker) server_var(gold_format)
      es_xforeachkey gold_key in "goldshop" "es_xdoblock wcs/addons/gold/checkkills"
    }
  }
}

block checkkills
{
  es_keygetvalue gold_kill goldshop server_var(gold_key) required
  es_keygetvalue gold_name goldshop server_var(gold_key) name
  if (server_var(gold_kills) = server_var(gold_kill)) do
  {
    es_delayed .5 es_tell event_var(attacker) #multi #default {GOLD} #lightgreenYou have unlocked a new GoldShop Item! #green server_var(gold_name)
    es_delayed .5 es_playsound event_var(attacker) "HOLLIDAY/gold/promoted.wav" 1
  }
}
