block goldadmin
{
  es_xgetcmduserid gold_userid
  es_getplayersteamid gold_steam server_var(gold_userid)
  es_exists gold_exists key goldadmin server_var(gold_steam)
  ifx true(gold_exists) do 
  { 
    es popup isqueued goldadmin x1 server_var(gold_userid)
    if (server_var(x1) = 0) then es popup send goldadmin server_var(gold_userid)
  }
  es_xelse do
  {
    es_tell server_var(gold_userid) #green You are not a Gold Admin
    es_playsound server_var(gold_userid) vo/citadel/br_laugh01.wav 1 server_var(gold_userid) 1
  }
}
