block +ability
{
  es_xgetcmduserid gold_use
  es_getplayersteamid gold_steam server_var(gold_use)
  es_keygetvalue gold_tmp golddata server_var(gold_steam) +ability

  {
    es_format gold_format "wcs/addons/gold/goldblocks/%1a" server_var(gold_tmp)
    es_doblock server_var(gold_format)

}
